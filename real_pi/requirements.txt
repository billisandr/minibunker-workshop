# real_pi — native (no-ROS, no-Docker) MiniBunker stack for the Raspberry Pi 5.

# Install into a venv on the Pi:
#   python3 -m venv ~/mb-venv --system-site-packages   # --system-site-packages
#                                                       # so picamera2 (apt) is visible
#   source ~/mb-venv/bin/activate
#   pip install -r requirements.txt
#
# picamera2 is NOT pip-installed here — on Raspberry Pi OS it ships via apt
# (python3-picamera2) tied to the system libcamera. The venv is created with
# --system-site-packages so the native camera path can import it. If picamera2
# is absent the camera falls back to a plain V4L2 cv2.VideoCapture.
