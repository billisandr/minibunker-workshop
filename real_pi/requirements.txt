# real_pi — native (no-ROS, no-Docker) MiniBunker stack for the Raspberry Pi 5.
#
# Install into a venv on the Pi:
#   python3 -m venv ~/mb-venv --system-site-packages   # --system-site-packages
#                                                       # so picamera2 (apt) is visible
#   source ~/mb-venv/bin/activate
#   pip install -r requirements.txt
#
# IMPORTANT (Raspberry Pi OS): numpy, OpenCV, PyYAML and picamera2 come from
# APT, not pip, and the venv is created with --system-site-packages so they are
# importable:
#     sudo apt install -y python3-numpy python3-opencv python3-yaml \
#                         python3-picamera2 can-utils
#     python3 -m venv ~/mb-venv --system-site-packages
# Do NOT `pip install numpy`/`opencv-python` on the Pi: the piwheels numpy wheel
# needs libopenblas and, installed into the venv, SHADOWS the working apt build
# (-> "libopenblas.so.0: cannot open shared object file"). picamera2 is likewise
# apt-only (tied to the system libcamera); absent it, the camera falls back to
# a plain V4L2 cv2.VideoCapture.
#
# So on the Pi, pip only needs python-can below. (On a non-Pi dev box without
# system numpy/cv2, also `pip install numpy opencv-python PyYAML`.)
python-can>=4.0           # socketcan driver for the Bunker CAN bus
